# 1. Create and activate an environment
python -m venv .venv

# Windows PowerShell
.venv\Scripts\Activate.ps1

# macOS/Linux
# source .venv/bin/activate

# 2. Install dependencies
python -m pip install --upgrade pip
pip install -r requirements.txt

# 3. Download the UCI Online Retail dataset
python download_data.py

# 4. Run the complete forecasting pipeline
python main.py

# 5. Optional: run the Streamlit dashboard
streamlit run app.py

# 6. Optional: install Ollama separately from its official website,
# then download the local Qwen3 model:
ollama pull qwen3:4b

# The forecasting pipeline does NOT depend on the LLM.
# If Ollama is unavailable, the numerical forecasting pipeline still completes.
